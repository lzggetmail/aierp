```python
def read_code_file(file_path):
    """读取Python代码文件"""
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            content = f.read()
        return {
            'success': True,
            'content': content,
            'lines': len(content.split('\n'))
        }
    except Exception as e:
        return {
            'success': False,
            'error': str(e)
        }
```

#### 初始方案：用AST

尝试用Python的ast模块分析代码：

**遇到的问题**：
- ❌ AST在Claude Code环境中不可用
- ❌ 需要额外依赖

#### 调整方案：用Grep + AI

改用Grep工具进行模式匹配：
```python
# 用Grep查找函数定义
functions = grep(code, pattern=r"^def \w+")

# 用Grep查找类定义
classes = grep(code, pattern=r"^class \w+")

# 用AI进行语义分析
analysis = claude.analyze(code)
```

**这个方案的优点**：
- ✅ 不需要额外依赖
- ✅ Grep快速可靠
- ✅ AI理解上下文

**实现代码**：
```python
def analyze_code_structure(code):
    """分析代码结构"""
    # 使用Grep查找模式
    functions = code.count("def ")
    classes = code.count("class ")

    # 使用Claude进行语义分析
    analysis = analyze_with_claude(code)

    return {
        'functions': functions,
```

**规则1：命名规范检查**

```python
def check_naming_convention(code):
    """检查命名规范"""
    issues = []

    # 检查函数名（应该是snake_case）
    if re.search(r'def [A-Z]', code):
        issues.append({
            'rule': 'R001',
            'severity': 'low',
            'message': '函数名应该使用snake_case',
            'suggestion': '将myFunction改为my_function'
        })

    return issues
```

**规则2：未使用的导入**
```python
def check_unused_imports(code):
    """检查未使用的导入"""
    imports = re.findall(r'^import \w+|^from \w+ import', code, re.M)

    issues = []
    for imp in imports:
        # 检查是否在代码中使用
        module = imp.split()[-1]
        if module not in code.split(imp)[1]:
            issues.append({
                'rule': 'R002',
                'severity': 'medium',
                'message': f'未使用的导入: {imp}',
                'suggestion': f'删除 {imp}'
            })

    return issues
```

**规则3：缺少文档字符串**
```python
def check_docstrings(code):
    """检查函数和类的文档字符串"""
    issues = []

    # 查找所有函数定义
    functions = re.finditer(r'def (\w+)\([^)]*\):', code)

    for match in functions:
        func_name = match.group(1)
        # 获取函数位置
        start = match.end()
        # 检查后面是否有文档字符串
        next_lines = code[start:start+100].strip()

        if not next_lines.startswith(('"""', "'''")):
            issues.append({
                'rule': 'R003',
                'severity': 'medium',
                'message': f'函数 {func_name} 缺少文档字符串',
                'suggestion': '添加函数说明文档'
            })

    return issues
```

**规则4：异常处理缺失**
```python
def check_exception_handling(code):
    """检查关键操作的异常处理"""
    issues = []

    # 文件操作
    file_ops = re.finditer(r'open\([^)]+\)', code)
    for match in file_ops:
        # 检查是否在try块中
        context = code[max(0, match.start()-100):match.end()+100]
        if 'try:' not in context:
            issues.append({
                'rule': 'R005',
                'severity': 'high',
                'message': '文件操作缺少异常处理',
                'suggestion': '使用try-except包裹文件操作',
                'example': '''
try:
    with open(file) as f:
        content = f.read()
except IOError as e:
    logger.error(f"无法读取文件: {e}")
                '''
            })

    return issues
```

- 参考了 [[正则表达式速查]]
