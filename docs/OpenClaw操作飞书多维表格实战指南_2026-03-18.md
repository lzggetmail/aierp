## 一、OpenClaw 与飞书集成方式

### 1.1 两种方式对比

| 方式 | 优势 | 劣势 | 适用场景 |
|-----|------|------|---------|
| **飞书官方插件** | 开箱即用、稳定可靠 | 功能有限 | 基础操作 |
| **自定义 Skill** | 灵活、可定制 | 需要开发 | 复杂业务逻辑 |

**建议**：两者结合使用，官方插件处理基础操作，自定义 Skill 处理业务逻辑。

---

### 1.2 架构图

```
用户指令（飞书 Chat）
    ↓
OpenClaw 接收
    ↓
意图识别
    ↓
调用 Skill
    ├── 官方插件：基础操作
    └── 自定义 Skill：业务逻辑
    ↓
Skill 操作 SQL Server（核心业务）
    ↓
Skill 同步结果到飞书多维表格
    ↓
返回结果给用户
```

## 二、飞书官方插件使用

### 2.1 安装飞书官方插件

```
方法 1：通过飞书妙搭一键部署
1. 访问飞书妙搭
2. 扫码或点击链接
3. OpenClaw 会自动安装飞书官方插件

方法 2：手动安装
在 OpenClaw 配置文件中添加：
npx skills add https://github.com/openclaw/feishu-plugin --skill feishu
```

---

### 2.2 飞书官方插件的能力

安装后，OpenClaw 会自动获得以下能力：

```
✅ 读取飞书上下文
├── 获取群聊消息
├── 获取文档内容
├── 获取日程安排
├── 获取任务列表
└── 获取多维表格数据

✅ 在飞书内执行操作
├── 发送消息
├── 写入文档
├── 创建/更新多维表格记录
├── 生成图表
└── 创建任务
```

---

### 2.3 配置飞书权限

确保 OpenClaw 有足够的权限：

```
需要申请的权限：
✅ bitable:app - 获取多维表格信息
✅ bitable:app:readonly - 读取多维表格
✅ bitable:app:write - 写入多维表格
✅ im:message - 发送消息
✅ im:chat - 读取聊天消息
```

## 三、编写自定义 Skill

### 3.1 Skill 开发框架

```javascript
// skill: feishu_bitable_operator

const lark = require('@larksuiteoapi/node-sdk');

// 初始化飞书客户端
function getFeishuClient() {
  return lark.Client.create({
    appId: process.env.FEISHU_APP_ID,
    appSecret: process.env.FEISHU_APP_SECRET
  });
}

// Skill 定义
const skill = {
  name: '飞书多维表格操作',
  description: '操作飞书多维表格，包括创建、查询、更新、删除记录',
  version: '1.0.0',

  // 操作函数
  functions: {
    // 创建记录
    createRecord: async (appToken, tableId, fields) => {
      const client = getFeishuClient();

      const result = await client.bitable.appTableRecord.create({
        path: {
          app_token: appToken,
          table_id: tableId
        },
        data: {
          fields: fields
        }
      });

      return result;
    },

    // 查询记录
    queryRecords: async (appToken, tableId, filter = '') => {
      const client = getFeishuClient();

      const params = {};
      if (filter) {
        params.filter = filter;
      }

      const result = await client.bitable.appTableRecord.list({
        path: {
          app_token: appToken,
          table_id: tableId
        },
        params: params
      });

      return result.data.items;
    },

    // 更新记录
    updateRecord: async (appToken, tableId, recordId, fields) => {
      const client = getFeishuClient();

      const result = await client.bitable.appTableRecord.update({
        path: {
          app_token: appToken,
          table_id: tableId,
          record_id: recordId
        },
        data: {
          fields: fields
        }
      });

      return result;
    },

    // 删除记录
    deleteRecord: async (appToken, tableId, recordId) => {
      const client = getFeishuClient();

      const result = await client.bitable.appTableRecord.delete({
        path: {
          app_token: appToken,
          table_id: tableId,
          record_id: recordId
        }
      });

      return result;
    }
  }
};

module.exports = skill;
```

## 四、实战示例

### 示例 1：查询产品库存

```javascript
// skill: query_inventory

/**
 * 查询产品库存
 *
 * 用户指令："帮我查一下 iPhone 15 Pro 的库存"
 */

const lark = require('@larksuiteoapi/node-sdk');

async function queryInventory(productName) {
  const client = lark.Client.create({
    appId: process.env.FEISHU_APP_ID,
    appSecret: process.env.FEISHU_APP_SECRET
  });

  // 从 SQL Server 查询
  const sql = require('mssql');
  await sql.connect({
    server: process.env.SQL_SERVER,
    database: process.env.SQL_DATABASE,
    user: process.env.SQL_USER,
    password: process.env.SQL_PASSWORD
  });

  const result = await sql.query`
    SELECT
      p.product_name,
      p.specification,
      i.quantity,
      w.warehouse_name
    FROM inventory i
    JOIN products p ON i.product_id = p.product_id
    JOIN warehouses w ON i.warehouse_id = w.warehouse_id
    WHERE p.product_name LIKE ${'%' + productName + '%'}
  `;

  // 如果有库存数据，同步到飞书
  if (result.recordset.length > 0) {
    await syncInventoryToFeishu(client, result.recordset);
  }

  // 返回结果
  return {
    success: true,
    data: result.recordset,
    message: `找到 ${result.recordset.length} 条库存记录`
  };
}

async function syncInventoryToFeishu(client, inventoryData) {
  const appToken = process.env.FEISHU_BITABLE_APP_TOKEN;
  const tableId = 'inventory'; // 库存表的 table_id

  // 批量写入飞书多维表格
  const records = inventoryData.map(item => ({
    fields: {
      '产品名称': item.product_name,
      '规格': item.specification,
      '仓库': item.warehouse_name,
      '库存数量': item.quantity,
      '更新时间': new Date().toISOString()
    }
  }));

  await client.bitable.appTableRecord.batchCreate({
    path: {
      app_token: appToken,
      table_id: tableId
    },
    data: {
      records: records
    }
  });
}

module.exports = {
  name: '查询库存',
  description: '查询产品库存并同步到飞书',
  function: queryInventory
};
```

### 示例 2：创建销售订单

```javascript
// skill: create_sales_order

/**
 * 创建销售订单
 *
 * 用户指令："帮王客户下一单，买 10 个 iPhone 15 Pro"
 */

const lark = require('@larksuiteoapi/node-sdk');
const sql = require('mssql');

async function createSalesOrder(params) {
  const { customerName, productName, quantity } = params;

  // 1. 连接 SQL Server
  await sql.connect({
    server: process.env.SQL_SERVER,
    database: process.env.SQL_DATABASE,
    user: process.env.SQL_USER,
    password: process.env.SQL_PASSWORD
  });

  // 2. 开启事务
  const transaction = new sql.Transaction();
  await transaction.begin();

  try {
    // 3. 查询客户信息
    const customerResult = await transaction.request()
      .input('customer_name', sql.NVarChar, customerName)
      .query('SELECT customer_id FROM customers WHERE customer_name = @customer_name');

    if (customerResult.recordset.length === 0) {
      throw new Error(`客户 "${customerName}" 不存在`);
    }

    const customerId = customerResult.recordset[0].customer_id;

    // 4. 查询产品信息
    const productResult = await transaction.request()
      .input('product_name', sql.NVarChar, productName)
      .query('SELECT product_id, price FROM products WHERE product_name = @product_name');

    if (productResult.recordset.length === 0) {
      throw new Error(`产品 "${productName}" 不存在`);
    }

    const productId = productResult.recordset[0].product_id;
    const price = productResult.recordset[0].price;

    // 5. 创建订单
    const orderResult = await transaction.request()
      .input('customer_id', sql.Int, customerId)
      .input('order_date', sql.DateTime, new Date())
      .input('total_amount', sql.Decimal(10, 2), price * quantity)
      .query(`
        INSERT INTO orders (customer_id, order_date, total_amount, status)
        OUTPUT INSERTED.order_id, INSERTED.order_code
        VALUES (@customer_id, @order_date, @total_amount, '待确认')
      `);

    const orderId = orderResult.recordset[0].order_id;
    const orderCode = orderResult.recordset[0].order_code;

    // 6. 创建订单明细
    await transaction.request()
      .input('order_id', sql.Int, orderId)
      .input('product_id', sql.Int, productId)
      .input('quantity', sql.Int, quantity)
      .input('price', sql.Decimal(10, 2), price)
      .input('amount', sql.Decimal(10, 2), price * quantity)
      .query(`
        INSERT INTO order_details (order_id, product_id, quantity, price, amount)
        VALUES (@order_id, @product_id, @quantity, @price, @amount)
      `);

    // 7. 提交事务
    await transaction.commit();

    // 8. 同步到飞书多维表格
    await syncOrderToFeishu({
      order_id: orderId,
      order_code: orderCode,
      customer_name: customerName,
      product_name: productName,
      quantity: quantity,
      amount: price * quantity
    });

    return {
      success: true,
      order_id: orderId,
      order_code: orderCode,
      message: `订单 ${orderCode} 创建成功，金额：¥${(price * quantity).toFixed(2)}`
    };

  } catch (error) {
    await transaction.rollback();
    throw error;
  }
}

async function syncOrderToFeishu(order) {
  const client = lark.Client.create({
    appId: process.env.FEISHU_APP_ID,
    appSecret: process.env.FEISHU_APP_SECRET
  });

  // 写入订单跟进表
  await client.bitable.appTableRecord.create({
    path: {
      app_token: process.env.FEISHU_BITABLE_APP_TOKEN,
      table_id: 'orders_followup'
    },
    data: {
      fields: {
        '订单编号': order.order_code,
        '客户名称': order.customer_name,
        '产品名称': order.product_name,
        '数量': order.quantity,
        '金额': order.amount,
        '状态': '待确认',
        '创建时间': new Date().toISOString()
      }
    }
  });
}

module.exports = {
  name: '创建销售订单',
  description: '创建销售订单并同步到飞书',
  function: createSalesOrder
};
```

### 示例 3：生成每日销售报告

```javascript
// skill: generate_daily_sales_report

/**
 * 生成每日销售报告
 *
 * 定时任务：每天晚上 10 点执行
 */

const lark = require('@larksuiteoapi/node-sdk');
const sql = require('mssql');

async function generateDailySalesReport(date) {
  // 1. 连接 SQL Server
  await sql.connect({
    server: process.env.SQL_SERVER,
    database: process.env.SQL_DATABASE,
    user: process.env.SQL_USER,
    password: process.env.SQL_PASSWORD
  });

  // 2. 查询当日销售数据
  const result = await sql.query`
    SELECT
      o.order_date,
      COUNT(*) as order_count,
      SUM(o.total_amount) as total_amount,
      (SELECT TOP 1 p.product_name
       FROM order_details od
       JOIN products p ON od.product_id = p.product_id
       WHERE od.order_id = o.order_id
       GROUP BY p.product_name
       ORDER BY SUM(od.amount) DESC) as top_product
    FROM orders o
    WHERE CAST(o.order_date AS DATE) = ${date}
    GROUP BY o.order_date
  `;

  if (result.recordset.length === 0) {
    return {
      success: false,
      message: `${date} 没有销售数据`
    };
  }

  const salesData = result.recordset[0];

  // 3. AI 分析
  const aiAnalysis = await analyzeSalesData(salesData);

  // 4. 写入飞书多维表格
  const client = lark.Client.create({
    appId: process.env.FEISHU_APP_ID,
    appSecret: process.env.FEISHU_APP_SECRET
  });

  await client.bitable.appTableRecord.create({
    path: {
      app_token: process.env.FEISHU_BITABLE_APP_TOKEN,
      table_id: 'sales_daily_report'
    },
    data: {
      fields: {
        '日期': date,
        '订单数量': salesData.order_count,
        '总销售额': salesData.total_amount,
        '热销产品': salesData.top_product,
        'AI 分析': aiAnalysis.insight,
        '建议措施': aiAnalysis.suggestion,
        '风险等级': aiAnalysis.riskLevel
      }
    }
  });

  // 5. 如果有异常，发送飞书消息通知
  if (aiAnalysis.riskLevel === '高') {
    await sendAlertToFeishu(client, aiAnalysis);
  }

  return {
    success: true,
    message: `销售报告已生成：${date}`
  };
}

async function analyzeSalesData(salesData) {
  // 这里可以调用 AI 模型进行分析
  // 简化示例：

  let insight = '';
  let suggestion = '';
  let riskLevel = '低';

  if (salesData.total_amount < 10000) {
    insight = `今日销售额 ${salesData.total_amount} 低于正常水平`;
    suggestion = '建议检查是否有订单遗漏或延迟';
    riskLevel = '中';
  } else {
    insight = `今日销售正常，热销产品：${salesData.top_product}`;
    suggestion = '继续保持，注意跟进待确认订单';
  }

  return { insight, suggestion, riskLevel };
}

async function sendAlertToFeishu(client, analysis) {
  await client.im.message.create({
    path: {
      receive_id_type: 'chat_id'
    },
    data: {
      receive_id: process.env.FEISHU_ALERT_CHAT_ID,
      msg_type: 'text',
      content: JSON.stringify({
        text: `⚠️ 销售预警\n${analysis.insight}\n建议：${analysis.suggestion}`
      })
    }
  });
}

module.exports = {
  name: '生成每日销售报告',
  description: '生成每日销售报告并同步到飞书',
  function: generateDailySalesReport
};
```

## 五、完整开发流程

### 5.1 环境准备

```bash
# 1. 安装 OpenClaw
npm install -g @openclaw/cli

# 2. 初始化 OpenClaw 项目
openclaw init my-aierp-agent

# 3. 安装飞书插件

npx skills add @openclaw/feishu
```

---

### 5.4 配置 OpenClaw

```javascript
// openclaw.config.js
module.exports = {
  name: 'AiERP Agent',
  description: 'AiERP 智能助手',

  // 飞书配置
  integrations: {
    feishu: {
      enabled: true,
      appId: process.env.FEISHU_APP_ID,
      appSecret: process.env.FEISHU_APP_SECRET
    }
  },

  // Skills 配置
  skills: [
    '@openclaw/feishu', // 官方插件
    './skills/query_inventory',
    './skills/create_sales_order',
    './skills/generate_daily_sales_report'
  ],

  // 定时任务
  schedules: [
    {
      cron: '0 22 * * *', // 每天晚上 10 点
      skill: 'generate_daily_sales_report',
      params: { date: new Date().toISOString().split('T')[0] }
    }
  ]
};
```

---

### 5.5 启动 OpenClaw

```bash
# 开发模式
openclaw dev

# 生产模式
openclaw start
```

```javascript
async function safeExecute(operation) {
  try {
    return await operation();
  } catch (error) {
    // 记录错误日志
```
